          # Extract Version from pubspec.yaml
          VERSION_LINE=$(grep -E '^version:' pubspec.yaml | sed 's/version:[[:space:]]*//' | tr -d '\r')
          VERSION=${VERSION_LINE%%+*}
          BUILD=${VERSION_LINE##*+}
          if [[ -z "$APP_NAME" || -z "$VERSION" ]]; then
            echo "::error::Could not parse name or version from pubspec.yaml"
            exit 1
          fi

          echo "APP_NAME=$APP_NAME" >> "$GITHUB_ENV"
          echo "BUILD_NAME=$VERSION" >> "$GITHUB_ENV"
          echo "BUILD_NUMBER=$BUILD" >> "$GITHUB_ENV"

          echo "name=$APP_NAME" >> "$GITHUB_OUTPUT"

      - name: Create or select iOS simulator
        run: |
          # Find the latest iOS runtime
          LATEST_IOS_RUNTIME=$(xcrun simctl list runtimes available | grep iOS | tail -1 | awk '{print $NF}' | tr -d '()')
          echo "Selected iOS runtime: $LATEST_IOS_RUNTIME"

          # Look for an existing iPhone 15 simulator with the selected runtime
          EXISTING_DEVICE=$(xcrun simctl list devices available | grep "iPhone 15" | grep "$LATEST_IOS_RUNTIME" | awk -F '[()]' '{print $2}' | head -1)

          if [ ! -z "$EXISTING_DEVICE" ]; then
            echo "Using existing iPhone 15 simulator: $EXISTING_DEVICE"
            echo "DEVICE_ID=$EXISTING_DEVICE" >> $GITHUB_ENV
          else
            echo "No existing iPhone 15 simulator found. Creating new one..."
            DEVICE_ID=$(xcrun simctl create "GitHub-Action-iPhone" "iPhone 15" "$LATEST_IOS_RUNTIME" || { echo "Failed to create simulator"; exit 1; })
            echo "Created simulator with ID: $DEVICE_ID"
            echo "DEVICE_ID=$DEVICE_ID" >> $GITHUB_ENV
          fi

        timeout-minutes: 3
        run: |
          # Force software rendering to avoid hardware acceleration issues

      UI.user_error!("Missing environment variable #{var} Please set it in your CI/CD pipeline or shell.") unless ENV[var]
    end
  end

  before_all do
    validate_env_vars!

    # Install CocoaPods dependencies
    UI.message("Installing CocoaPods dependencies...")
    cocoapods(
      clean_install: true,
      podfile: PODFILE_PATH
    )
    UI.success("CocoaPods dependencies installed.")
  end

  private_lane :clean_builds do
    UI.message("Cleaning builds...")
    sh "flutter clean"
    sh "flutter pub get"
  end

  private_lane :scan_for_secrets do
    # git-secrets is used to detect secrets
    sh "git secrets --scan"
    UI.message("No secrets detected in code")
  rescue => e
    UI.user_error!("Secrets detected in code: #{e.message}")
  end

  private_lane :audit_dependencies do
    UI.message("Auditing Flutter dependencies...")

    # Check unknown vulnerabilities
    sh "flutter pub deps --style=compact"
  end

  private_lane :check_dependencies do
    UI.message("Checking dependencies...")

    sh "flutter pub get"

    # Verify dependency integrity
    sh "dart analyze --fatal-infos ."

    UI.message("Dependency status checked")
  end

  private_lane :run_code_analysis do
    UI.message("Running code analysis...")

    sh "flutter analyze"

    sh "dart analyze --fatal-infos ."

    UI.message("Code analysis passed")
  end

  private_lane :run_tests_with_coverage do
    UI.message("Running tests with coverage...")
    sh "flutter test --coverage"

    UI.message("Tests passed")
  end

  private_lane :flutter_test do
    run_tests_with_coverage
  end

  # Private lane to build iOS app with Flutter
  # Options:
  # :flavor (String): The Flutter flavor (e.g., "prod", "dev"). Defaults to "prod"
  # :export_method (String): The Xcode export method (e.g., "development", "app-store", "ad-hoc", "enterprise"). Defaults to "development"
  # :build_number (String): The build number for the archive. Required
  private_lane :flutter_build_ios do |options|
    clean_builds

    flavor = options[:flavor] || "prod"
    export_method = options[:export_method] || "development"
    build_number = options[:build_number] || BUILD_NUMBER
    scheme = (flavor == "prod") ? "Runner" : "Runner-#{flavor}"

    # unless ENV["APP_IDENTIFIER"] && ENV["PROVISIONING_PROFILE_NAME"]
    #   UI.user_error!("APP_IDENTIFIER and PROVISIONING_PROFILE_NAME must be set in your environment variables.")
    # end

    UI.message("Building Flutter iOS app with flavor #{flavor}, export method: #{export_method}, Build number: #{build_number}...")
    # Build the Flutter iOS app (no codesign as gym will handle it)
    sh("flutter build ios --#{flavor} --release --obfuscate --split-debug-info=ios/debug-info --dart-define-from-file=.env --build-number=#{build_number} --no-codesign")

    # Build the Xcode archive using gym
    gym(
      workspace: XCODE_WORKSPACE_PATH,
      scheme: scheme,
      clean: true,
      export_method: export_method,
      output_directory: OUTPUT_DIR,
      output_name: "#{flavor}_#{build_number}.ipa", # Include build number in output name
      # When using match, gym usually picks up the correct provisioning profile automatically
      # You can explicitly specify provisioningProfiles if needed, but match simplifies this.
      export_options: {
        provisioningProfiles: {
          ENV["APP_IDENTIFIER"] => ENV["PROVISIONING_PROFILE_NAME"]
          # e.g., "com.yourcompany.yourapp" => "match AppStore com.yourcompany.yourapp"
        }
      }
    )

    UI.success("iOS app built successfully. to #{File.join(OUTPUT_DIR, "#{flavor}_#{build_number}.ipa")}")
  end

  # Build the iOS app in debug mode
  desc "Build the iOS app in debug (development) mode"
  lane :build_debug do
    current_build_number = latest_testflight_build_number(xcodeproj: XCODEPROJ_PATH) + 1
    UI.message("Debug build: current build number set to #{current_build_number}")

    # Ensure development match profiles are available if not done before_all
    match(type: "development", git_url: ENV["MATCH_GIT_URL"], readonly: is_ci?)

    flutter_build_ios(
      flavor: "dev",
      export_method: "development",
      build_number: current_build_number
    )
  end

  # Build the iOS app in release mode
  desc "Build the iOS app in release (App Store) mode"
  lane :build_release do
    current_build_number = latest_testflight_build_number(xcodeproj: XCODEPROJ_PATH) + 1
    UI.message("Release build: current build number set to #{current_build_number}")

    # Ensure appstore match profiles are available if not done before_all
    match(type: "appstore", git_url: ENV["MATCH_GIT_URL"], readonly: is_ci?)

    flutter_build_ios(
      flavor: "prod",
      export_method: "app-store",
      build_number: current_build_number
    )
  end

  desc "Submit a new beta build to TestFlight"
  lane :beta do
    # Get the latest build number and increment it for the new Build
    # Handle the case where there are no previous TestFlight builds
    begin
      next_build_number = latest_testflight_build_number(xcodeproj: XCODEPROJ_PATH) + 1
    rescue
      UI.warning("Could not determine latest TestFlight build number: #{next_build_number}")
      next_build_number = 1 # Start from 1 if no existing builds are found
    end

    UI.message("Preparing to submit to TestFlight with build number: #{next_build_number}")

    # Increment the build number in Xcode project (optional can be dione by gym/flutter build if passed correctly)
    # This ensures the .xcodeproj file also reflects the build number.

    increment_build_number(
      build_number: next_build_number,
      xcodeproj: XCODEPROJ_PATH
    )

    # Build the release version of the app
    # The `flutter_build_ios` lane will use the `next_build_number`
    flutter_build_ios(
      flavor: "prod",
      export_method: "app-store", # TestFlight builds usually use "app-store" export method
      build_number: next_build_number
    )

    # Upload to TestFlight
    upload_to_testflight(
      skip_waiting_for_build_processing: true, # Speeds up the lane execution
      # TODO:
      apple_id: ENV["APPLE_ID"], # e.g., "your_apple_id@email.com"
      team_id: ENV["TEAM_ID"] # "YOUR_TEAM_ID"
      # groups: ["testers"], Optional: Specify tester groups
      # changelog: "Beta build for testing." # Optional: Add a changelog message
    )
    UI.success("Successfully uploaded new beta build to TestFlight!")
  end

  # Deploy to App Store
  desc "Deploy a new version to the App Store"
  lane :deploy do
    # TODO: `fastlane deliver` to upload metadata
    # Increment version number (you might want to use semantic versioning)
    increment_version_number(
      version_number: BUILD_NAME, # TODO: Replace with your versioning logic
      xcodeproj: XCODEPROJ_PATH
    )
    # Build the release
    build_release
    # Upload to the App Store Connect
    upload_to_app_store(
      skip_metadata: true, # TODO: 
      skip_screenshots: true,
      apple_id: ENV["APPLE_ID"], # e.g., "your_apple_id@email.com"
      team_id: ENV["TEAM_ID"] # "YOUR_TEAM_ID"
    )
  end

    # add actions here: https://docs.fastlane.tools/actions
  end

    puts "Fastlane finished successfully!"
